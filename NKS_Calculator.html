<script>
    // Load the data from the JSON files (We will load these next step)
    let miamiData = [];
    let chicagoData = [];
    let laData = [];

    async function loadData() {
        try {
            const miamiResp = await fetch('miami_nks.json');
            miamiData = await miamiResp.json();
            const chicagoResp = await fetch('chicago_nks.json');
            chicagoData = await chicagoResp.json();
            const laResp = await fetch('la_nks.json');
            laData = await laResp.json();
            console.log("✅ Data loaded successfully!");
        } catch (e) {
            console.log("⚠️ Data files not found locally. Please ensure JSON files are in the same folder as this HTML.");
        }
    }

    // Load data when the page opens
    loadData();

    function calculateNKS() {
        const city = document.getElementById('citySelect').value;
        const year = parseInt(document.getElementById('yearInput').value);
        const resultsDiv = document.getElementById('results');
        const warningMsg = document.getElementById('warningMsg');
       
        resultsDiv.style.display = 'none';
        warningMsg.style.display = 'none';

        if (!city || !year) {
            alert("Please select a city and enter a year.");
            return;
        }

        let data = [];
        if (city === 'miami') data = miamiData;
        else if (city === 'chicago') data = chicagoData;
        else if (city === 'la') data = laData;

        // Find the entry for the specific year
        const entry = data.find(item => item.Year === year);

        if (entry) {
            // Display results
            document.getElementById('outputOmega').innerText = entry.Omega_NKS.toFixed(3);
            // Calculate savings based on Omega (simplified)
            let savings = (1 - 1/entry.Omega_NKS) * 100;
            document.getElementById('outputSavings').innerText = savings.toFixed(1) + '%';
            document.getElementById('outputLoad').innerText = entry.V_50.toFixed(2) + ' m/s';
            resultsDiv.style.display = 'block';
        } else {
            warningMsg.style.display = 'block';
        }
    }
</script>
